    const people=eligible.filter(m=>memberRoles(m).some(r=>cfg.roles.includes(r)));

  });

  membersEl.innerHTML=groups.length?groups.map((g,idx)=>`<div class="flow-group"><div class="flow-group-head"><div class="flow-step-no">${idx+1}</div><div><div class="flow-role-title">${g.title}</div><div class="flow-role-sub">${g.note}</div></div></div><div class="flow-members">${g.people.map(routePersonCard).join('')}</div></div>${idx<groups.length-1?'<div class="flow-connector">↓</div>':''}`).join(''):'<div class="matched-empty">目前尚未設定可處理這類問題的成員。</div>';
