async function loadBackendProfiles(){

    publicSb.from('bni_need_members').select('*').order('sort_order'),
    publicSb.from('bni_need_step_members').select('*').order('step_index').order('sort_order')
  ]);
  if(mr.error){console.warn('Supabase 成員讀取失敗',mr.error);return}
  const members=mr.data||[],cases=cr.data||[],collabs=br.data||[],contacts=tr.data||[];
  const byId=new Map(members.map(m=>[m.id,m]));backendProfiles.clear();
  members.forEach(m=>backendProfiles.set(m.name,{member:m,cases:[],collabs:[],contacts:[]}));

  contacts.forEach(x=>{const m=byId.get(x.member_id);if(m&&backendProfiles.has(m.name))backendProfiles.get(m.name).contacts.push(x)});
  const mapped=members.map(m=>({id:m.id,company:m.company||'',name:m.name,desc:m.tagline||'',expertise_category:m.expertise_category||'',services:m.services||'',tags:memberProblems(m),problem_keys:memberProblems(m),role_keys:memberRoles(m)}));allMembers.splice(0,allMembers.length,...mapped);

function renderProfessionalDetails(row){
  const section=document.getElementById('professionalSection'),wrap=document.getElementById('professionalDetails');
  if(!section||!wrap)return;
  const category=String(row.expertise_category||'').trim(),services=String(row.services||'').trim(),problems=String(row.common_problems||'').trim(),scope=String(row.work_scope||'').trim();
  const hasAny=!!(category||services||problems||scope);
  section.classList.toggle('is-hidden',!hasAny);
  if(!hasAny){wrap.innerHTML='';return}
  const esc=v=>String(v).replace(/[&<>"']/g,ch=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[ch]));
  const tags=services?services.split(/[、，,；;]/).map(x=>x.trim()).filter(Boolean):[];
  const hasFlow=/→|->|＞/.test(scope);
  const flow=hasFlow?scope.split(/\s*(?:→|->|＞)\s*/).map(x=>x.trim()).filter(Boolean):[];
  const categoryHtml=category?`<div class="professional-category">${esc(category)}</div>`:'';
  const serviceHtml=tags.length?`<div class="professional-block"><div class="professional-detail-label">主要服務</div><div class="professional-tags">${tags.map(x=>`<span class="professional-tag">${esc(x)}</span>`).join('')}</div></div>`:'';
  const problemHtml=problems?`<div class="professional-block"><div class="professional-detail-label">擅長處理</div><div class="professional-text">${esc(problems)}</div></div>`:'';
  const scopeHtml=scope?(hasFlow?`<div class="professional-block"><div class="professional-detail-label">服務流程</div><div class="professional-flow">${flow.map((x,i)=>`${i?'<span class="professional-flow-arrow">→</span>':''}<span class="professional-flow-step">${esc(x)}</span>`).join('')}</div></div>`:`<div class="professional-block"><div class="professional-detail-label">常見施工／服務內容</div><div class="professional-text">${esc(scope)}</div></div>`):'';
  wrap.innerHTML=categoryHtml+serviceHtml+problemHtml+scopeHtml;
}
