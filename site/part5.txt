<script>
const PUBLIC_SB_URL='https://packaiwgsswrjvxdsejw.supabase.co';
const PUBLIC_SB_KEY='sb_publishable_Jrp7cxW-ppSJtrk-lScZjA_blNEnBRG';
const publicSb=supabase.createClient(PUBLIC_SB_URL,PUBLIC_SB_KEY,{auth:{persistSession:false}});
const backendProfiles=new Map();
const existingTagMap=new Map(allMembers.map(m=>[m.name,m.tags||['unknown']]));

function contactIcon(type){return type==='phone'?'☎':type==='instagram'?'IG':type==='facebook'?'f':String(type||'').includes('line')?'LINE':'↗'}
function renderDirectoryFromBackend(){
  const grid=document.querySelector('.directory-grid'); if(!grid)return;
  grid.innerHTML=allMembers.map(m=>`<div class="member directory-card"><div class="company">${m.company||''}</div><div class="name">${m.name}</div><div class="desc">${m.desc||''}</div></div>`).join('');
  syncDirectoryCardsInteractivity();
}
async function loadBackendProfiles(){
  const [mr,cr,br,tr]=await Promise.all([
    publicSb.from('bni_members').select('*').eq('is_published',true).order('sort_order'),
    publicSb.from('bni_member_cases').select('*').eq('is_published',true).order('sort_order'),
    publicSb.from('bni_member_collaborations').select('*').eq('is_published',true).order('sort_order'),
    publicSb.from('bni_member_contacts').select('*').eq('is_published',true).order('sort_order')
  ]);
  if(mr.error){console.warn('Supabase 成員讀取失敗',mr.error);return}
  const members=mr.data||[], cases=cr.data||[], collabs=br.data||[], contacts=tr.data||[];
  const byId=new Map(members.map(m=>[m.id,m]));
  backendProfiles.clear();
  members.forEach(m=>backendProfiles.set(m.name,{member:m,cases:[],collabs:[],contacts:[]}));
  cases.forEach(x=>{const m=byId.get(x.member_id);if(m&&backendProfiles.has(m.name))backendProfiles.get(m.name).cases.push(x)});
  collabs.forEach(x=>{const m=byId.get(x.member_id);if(m&&backendProfiles.has(m.name))backendProfiles.get(m.name).collabs.push(x)});
  contacts.forEach(x=>{const m=byId.get(x.member_id);if(m&&backendProfiles.has(m.name))backendProfiles.get(m.name).contacts.push(x)});

  const mapped=members.map(m=>({company:m.company||'',name:m.name,desc:m.tagline||'',tags:existingTagMap.get(m.name)||['unknown']}));
  allMembers.splice(0,allMembers.length,...mapped);
  members.forEach(m=>{
    if(m.card_json_url){discoveredCardJsonUrls[m.name]=m.card_json_url;enabledProfiles.add(m.name)}
    const p=backendProfiles.get(m.name);
    if((m.intro||'').trim()||p.cases.length||p.collabs.length||p.contacts.length)enabledProfiles.add(m.name);
  });
  renderDirectoryFromBackend();
  renderMatchedMembers(currentNeedKey);
}

renderContacts=async function(m){
  const wrap=document.getElementById('profileContacts'),p=backendProfiles.get(m.name),rows=p?.contacts||[];
  if(!rows.length){wrap.innerHTML='';document.getElementById('contactSection').classList.add('is-hidden');return}
  document.getElementById('contactSection').classList.remove('is-hidden');
  wrap.innerHTML=rows.map(c=>`<a class="contact-action" href="${c.url}" target="_blank" rel="noopener noreferrer"><div class="contact-icon">${contactIcon(c.type)}</div><div class="contact-copy"><div class="contact-type">${c.label||''}</div><div class="contact-display">${c.display_value||''}</div></div></a>`).join('');
};

openProfile=async function(name){
  if(!hasProfileData(name))return;
  const m=memberByName(name); if(!m)return;
  const p=backendProfiles.get(name)||{member:{},cases:[],collabs:[],contacts:[]};
  const row=p.member||{};
  document.getElementById('profileCompany').textContent=m.company||'';
  document.getElementById('profileName').textContent=m.name||'';
  document.getElementById('profileTagline').textContent=m.desc||'';
  await renderEcardGallery(m);
  const intro=(row.intro||'').trim(), hasIntro=!!intro, hasCases=p.cases.length>0, hasBni=p.collabs.length>0;
  const introSection=document.getElementById('introSection'),caseSection=document.getElementById('caseSection'),bniSection=document.getElementById('bniSection'),profileGrid=document.querySelector('.profile-grid');
  introSection.classList.toggle('is-hidden',!hasIntro);caseSection.classList.toggle('is-hidden',!hasCases);bniSection.classList.toggle('is-hidden',!hasBni);profileGrid.classList.toggle('single-column',!hasBni);
  document.getElementById('profileIntro').textContent=intro;
  const caseEl=document.getElementById('profileCases');
  caseEl.classList.toggle('three-cases',p.cases.length===3);
  caseEl.innerHTML=hasCases?p.cases.map(c=>`${c.link_url?`<a class="case-link" href="${c.link_url}" target="_blank" rel="noopener noreferrer">`:''}<div class="case-item ${c.link_url?'case-clickable':''}">${c.image_url?`<img src="${c.image_url}" alt="${c.title||''}" style="width:100%;aspect-ratio:4/3;object-fit:cover;border-radius:10px;margin-bottom:12px">`:''}<b>${c.title||''}</b><span>${c.description||''}</span>${c.link_url?'<span class="case-source">查看案例 →</span>':''}</div>${c.link_url?'</a>':''}`).join(''):'';
  document.getElementById('profileBNI').innerHTML=hasBni?p.collabs.map(c=>`<div class="bni-item">${c.image_url?`<img src="${c.image_url}" alt="${c.title||''}" style="width:100%;max-height:240px;object-fit:cover;border-radius:10px;margin-bottom:10px">`:''}<b>${c.title||''}</b><span>${[c.partner_name,c.collaboration_date].filter(Boolean).join('｜')}</span><span>${c.description||''}</span>${c.link_url?`<a href="${c.link_url}" target="_blank" rel="noopener noreferrer" class="case-source">查看內容 →</a>`:''}</div>`).join(''):'';
  await renderContacts(m);
  document.getElementById('profileOverlay').classList.add('open');document.body.style.overflow='hidden';
};

window.addEventListener('DOMContentLoaded',async()=>{await loadBackendProfiles()});
