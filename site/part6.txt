<script>
(function(){
  const labelMap={
    'introSection':'介紹',
    'profile-card-showcase':'電子名片',
    'bniSection':'BNI合作',
    'caseSection':'案例',
    'contactSection':'聯絡方式'
  };
  let activeTarget='';
  let scrollTick=false;

  function getSectionElement(id){
    return id==='profile-card-showcase'?document.querySelector('.profile-card-showcase'):document.getElementById(id);
  }
  function getVisibleSections(){
    return Object.keys(labelMap)
      .map(id=>({id,label:labelMap[id],el:getSectionElement(id)}))
      .filter(x=>x.el&&!x.el.classList.contains('is-hidden'))
      .sort((a,b)=>{
        if(a.el===b.el)return 0;
        const pos=a.el.compareDocumentPosition(b.el);
        return pos&Node.DOCUMENT_POSITION_FOLLOWING?-1:1;
      });
  }
  function syncStickyOffsets(){
    const panel=document.querySelector('.profile-panel'),hero=document.querySelector('.profile-hero'),nav=document.getElementById('mobileProfileNav');
    if(!panel||!hero||!nav)return;
    panel.style.setProperty('--profile-hero-h',`${hero.offsetHeight}px`);
    panel.style.setProperty('--profile-nav-h',`${nav.offsetHeight}px`);
  }
  function setActiveTab(id){
    activeTarget=id||'';
    const nav=document.getElementById('mobileProfileNav');if(!nav)return;
    [...nav.querySelectorAll('button[data-target]')].forEach(b=>b.classList.toggle('active',b.dataset.target===activeTarget));
  }
  function resetProfilePosition(){
    const panel=document.querySelector('.profile-panel');
    const gallery=document.getElementById('ecardGallery');
    if(panel)panel.scrollTop=0;
    if(gallery)gallery.scrollLeft=0;
    const first=getVisibleSections()[0];
    setActiveTab(first?.id||'');
    requestAnimationFrame(()=>{syncStickyOffsets();updateActiveFromScroll()});
  }
  function updateActiveFromScroll(){
    const panel=document.querySelector('.profile-panel'),hero=document.querySelector('.profile-hero'),nav=document.getElementById('mobileProfileNav');
    if(!panel||!hero||!nav)return;
    const sections=getVisibleSections();if(!sections.length)return;
    if(panel.scrollTop+panel.clientHeight>=panel.scrollHeight-24){setActiveTab(sections[sections.length-1].id);return}
    const threshold=panel.getBoundingClientRect().top+hero.offsetHeight+nav.offsetHeight+18;
    let current=sections[0];
    for(const s of sections){
      if(s.el.getBoundingClientRect().top<=threshold)current=s;
      else break;
    }
    setActiveTab(current.id);
  }
  function ensureMobileUi(){
    const body=document.querySelector('.profile-body');
    const panel=document.querySelector('.profile-panel');
    if(!body||!panel)return;
    let nav=document.getElementById('mobileProfileNav');
    if(!nav){
      nav=document.createElement('div');
      nav.id='mobileProfileNav';nav.className='mobile-profile-nav';
      panel.insertBefore(nav,body);
      nav.addEventListener('click',e=>{
        const b=e.target.closest('button[data-target]');if(!b)return;
        const t=getSectionElement(b.dataset.target);if(!t)return;
        setActiveTab(b.dataset.target);
        t.scrollIntoView({behavior:'smooth',block:'start'});
      });
    }else if(nav.parentElement!==panel){
      panel.insertBefore(nav,body);
    }
    if(!panel.dataset.mobileScrollSpy){
      panel.dataset.mobileScrollSpy='1';
      panel.addEventListener('scroll',()=>{
        if(scrollTick)return;
        scrollTick=true;
        requestAnimationFrame(()=>{scrollTick=false;updateActiveFromScroll()});
      },{passive:true});
    }
    let bar=document.getElementById('mobileContactBar');
    if(!bar){bar=document.createElement('div');bar.id='mobileContactBar';bar.className='mobile-contact-bar';document.body.appendChild(bar)}
    refreshMobileUi();
  }
  function refreshNav(){
    const nav=document.getElementById('mobileProfileNav');if(!nav)return;
    const items=getVisibleSections();
    const validActive=items.some(x=>x.id===activeTarget)?activeTarget:(items[0]?.id||'');
    activeTarget=validActive;
    nav.innerHTML=items.map(x=>`<button type="button" data-target="${x.id}" class="${x.id===validActive?'active':''}">${x.label}</button>`).join('');
    requestAnimationFrame(()=>{syncStickyOffsets();updateActiveFromScroll()});
  }
  function refreshCounter(){
    const gallery=document.getElementById('ecardGallery');if(!gallery)return;
    let counter=document.getElementById('ecardCounter');
    if(!counter){counter=document.createElement('div');counter.id='ecardCounter';counter.className='ecard-counter';gallery.insertAdjacentElement('afterend',counter)}
    const pages=[...gallery.querySelectorAll('.ecard-page')];
    if(!pages.length){counter.textContent='';return}
    const update=()=>{const i=Math.max(0,Math.min(pages.length-1,Math.round(gallery.scrollLeft/Math.max(1,gallery.clientWidth))));counter.textContent=`${i+1} / ${pages.length}`};
    gallery.onscroll=()=>requestAnimationFrame(update);update();
  }
  function refreshBottomContacts(){
    const bar=document.getElementById('mobileContactBar'),src=document.getElementById('profileContacts');if(!bar||!src)return;
    const links=[...src.querySelectorAll('a')].slice(0,3);
    bar.innerHTML=links.map(a=>`<a href="${a.getAttribute('href')||'#'}" target="_blank" rel="noopener noreferrer">${(a.querySelector('.contact-type')?.textContent||'聯絡').trim()}</a>`).join('');
    bar.classList.toggle('has-items',links.length>0);
  }
  function refreshMobileUi(){refreshNav();refreshCounter();refreshBottomContacts();requestAnimationFrame(syncStickyOffsets)}
  window.addEventListener('DOMContentLoaded',()=>{
    ensureMobileUi();
    const overlay=document.getElementById('profileOverlay');
    const gallery=document.getElementById('ecardGallery');
    const contacts=document.getElementById('profileContacts');
    const profileName=document.getElementById('profileName');
    if(overlay)new MutationObserver(mutations=>{
      const meaningful=mutations.some(m=>{
        const t=m.target;
        return !(t&&t.closest&&t.closest('#mobileProfileNav'));
      });
      if(meaningful)setTimeout(refreshMobileUi,50);
    }).observe(overlay,{attributes:true,subtree:true,childList:true,attributeFilter:['class']});
    if(profileName)new MutationObserver(()=>{
      activeTarget='';
      setTimeout(()=>{refreshMobileUi();resetProfilePosition()},0);
      setTimeout(resetProfilePosition,80);
    }).observe(profileName,{childList:true,characterData:true,subtree:true});
    if(gallery)new MutationObserver(()=>setTimeout(refreshCounter,30)).observe(gallery,{childList:true,subtree:true});
    if(contacts)new MutationObserver(()=>setTimeout(refreshBottomContacts,30)).observe(contacts,{childList:true,subtree:true});
    window.addEventListener('resize',()=>requestAnimationFrame(()=>{syncStickyOffsets();updateActiveFromScroll()}));
  });
})();
</script>
