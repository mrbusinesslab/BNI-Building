<script>
(function(){
  let currentIntent='';
  const intentButtons=[['找人','person_search'],['找專業','profession_search'],['解決問題','problem_solve'],['我不知道該找誰','guided_help']];
  function getMembers(){return typeof allMembers!=='undefined'&&Array.isArray(allMembers)?allMembers:[]}
  function getNeeds(){return typeof needsData!=='undefined'&&Array.isArray(needsData)?needsData.filter(n=>n&&n.key&&n.key!=='unknown'):[]}
  function getBackendProfile(name){return typeof backendProfiles!=='undefined'&&backendProfiles?.get?backendProfiles.get(name):null}
  function getStepRows(key){return typeof needStepMembers!=='undefined'&&needStepMembers?.get?(needStepMembers.get(key)||[]):[]}
  function track(event_type,data={}){try{if(typeof publicSb==='undefined'||!publicSb)return;publicSb.from('bni_behavior_events').insert({event_type,member_name:data.member_name||null,intent:data.intent||null,query_text:data.query_text||null,meta:data.meta||{}}).then(()=>{})}catch(e){}}
  function addMsg(text,who='bot'){const body=document.getElementById('assistantBody');if(!body)return;const d=document.createElement('div');d.className='assistant-msg '+who;d.textContent=text;body.appendChild(d);body.scrollTop=body.scrollHeight}
  function addChoices(items,attr){const body=document.getElementById('assistantBody');if(!body)return;const box=document.createElement('div');box.className='assistant-choices';box.innerHTML=items.map(([label,key])=>`<button class="assistant-choice" ${attr}="${key}">${label}</button>`).join('');body.appendChild(box);body.scrollTop=body.scrollHeight}
  function showChoices(){addChoices(intentButtons,'data-intent')}
  function showGuidedNeeds(){const needs=getNeeds();if(!needs.length){addMsg('目前沒有開放中的需求類別。');return}addMsg('你目前比較接近哪一種需求？');addChoices(needs.map(n=>[n.title,n.key]),'data-need')}
  function showFallback(){addMsg('我目前還無法判斷你的需求。你可以換個方式描述，或直接從目前開放的需求類別選擇。');showGuidedNeeds();track('chat_fallback',{intent:currentIntent})}
  function memberSearchText(m){const p=getBackendProfile(m.name)?.member||{};return [m.name,m.company,m.desc,p.title,p.expertise_category,p.services,p.common_problems,p.work_scope,p.tagline].filter(Boolean).join(' ').toLowerCase()}
  function showMembers(list,prefix){const visible=getMembers(),visibleIds=new Set(visible.map(m=>m.id)),uniq=[];const seen=new Set();list.forEach(m=>{if(m&&visibleIds.has(m.id)&&!seen.has(m.id)){seen.add(m.id);uniq.push(m)}});if(!uniq.length){showFallback();return}addMsg(prefix||`找到 ${uniq.length} 位可能適合的成員：`);const body=document.getElementById('assistantBody');uniq.slice(0,8).forEach(m=>{const d=document.createElement('div');d.className='assistant-member';d.innerHTML=`<b>${m.name}｜${m.company||''}</b><span>${m.desc||''}</span><button data-member="${m.name}">查看人物</button>`;body.appendChild(d)});body.scrollTop=body.scrollHeight}
  function findPeople(q){const x=q.toLowerCase();return getMembers().filter(m=>String(m.name||'').toLowerCase().includes(x)||String(m.company||'').toLowerCase().includes(x))}
  function findProfession(q){const x=q.toLowerCase().replace(/\s+/g,'');const direct=getMembers().filter(m=>memberSearchText(m).replace(/\s+/g,'').includes(x));if(direct.length)return direct;const tokens=[...new Set(q.toLowerCase().split(/[\s、，,。\/｜]+/).map(s=>s.trim()).filter(s=>s.length>=2&&!['我要','我想','幫我','請問','找人','找專業'].includes(s)))];if(!tokens.length)return[];return getMembers().map(m=>({m,score:tokens.reduce((n,t)=>n+(memberSearchText(m).includes(t)?1:0),0)})).filter(x=>x.score>0).sort((a,b)=>b.score-a.score).map(x=>x.m)}
  function membersForNeed(key,stepIndex=null){const members=getMembers(),byId=new Map(members.map(m=>[m.id,m]));let rows=getStepRows(key);if(stepIndex!==null)rows=rows.filter(r=>Number(r.step_index)===Number(stepIndex));rows=[...rows].sort((a,b)=>(a.step_index-b.step_index)||((a.sort_order||0)-(b.sort_order||0)));const out=[],seen=new Set();rows.forEach(r=>{const m=byId.get(r.member_id);if(m&&!seen.has(m.id)){seen.add(m.id);out.push(m)}});return out}
  function scoreNeed(q,n){const x=q.toLowerCase();let score=0;const title=String(n.title||'').toLowerCase(),small=String(n.small||'').toLowerCase(),text=String(n.text||'').toLowerCase();if(title&&x.includes(title))score+=12;if(small&&x.includes(small))score+=4;const steps=Array.isArray(n.steps)?n.steps:[];steps.forEach(s=>{const parts=String(s).toLowerCase().split(/[／/、與和\s]+/).filter(w=>w.length>=2);parts.forEach(w=>{if(x.includes(w))score+=5})});const words=(title+' '+small+' '+text).split(/[\s、，,。／/｜]+/).filter(w=>w.length>=2);words.forEach(w=>{if(x.includes(w))score+=2});return score}
  function bestNeedMatch(q){return getNeeds().map(n=>({n,score:scoreNeed(q,n)})).filter(x=>x.score>0).sort((a,b)=>b.score-a.score)[0]?.n||null}
  function bestStepIndex(q,need){const x=q.toLowerCase(),steps=Array.isArray(need?.steps)?need.steps:[];let best=-1,bestScore=0;steps.forEach((s,i)=>{let score=0;String(s).toLowerCase().split(/[／/、與和\s]+/).filter(w=>w.length>=2).forEach(w=>{if(x.includes(w))score++});if(score>bestScore){bestScore=score;best=i}});return bestScore?best:null}
  function recommendByProblem(q){const need=bestNeedMatch(q);if(!need)return false;const step=bestStepIndex(q,need);let list=membersForNeed(need.key,step);if(!list.length)list=membersForNeed(need.key);if(!list.length)return false;const stepName=step!==null?need.steps?.[step]:'';showMembers(list,stepName?`依你描述的狀況，較接近「${need.title}－${stepName}」：`:`依你描述的狀況，較接近「${need.title}」：`);track('chat_problem_pick',{intent:need.key,query_text:q,meta:{need_key:need.key,step_index:step}});return true}
  function handleIntent(key){currentIntent=key;track('chat_intent',{intent:key});if(key==='person_search')addMsg('請輸入姓名或公司名稱。');else if(key==='profession_search')addMsg('請輸入你要找的專業，例如：燈光、水電、窗簾、辦公家具。');else if(key==='problem_solve')addMsg('請直接描述你遇到的工程或空間問題。');else showGuidedNeeds()}
  function handleNeed(key){const need=getNeeds().find(n=>n.key===key);if(!need)return;currentIntent='problem_solve';const list=membersForNeed(key);showMembers(list,`「${need.title}」目前可從這些專業成員開始：`);track('chat_problem_pick',{intent:key,meta:{need_key:key}});if(typeof selectNeed==='function')selectNeed(key,false)}
  function handleText(q){q=q.trim();if(!q)return;addMsg(q,'user');track('chat_message',{intent:currentIntent,query_text:q});if(currentIntent==='person_search'){const list=findPeople(q);return list.length?showMembers(list,'我找到這些成員：'):showFallback()}const people=findPeople(q);if(people.length)return showMembers(people,'我找到你提到的人物：');if(currentIntent==='profession_search'){const list=findProfession(q);return list.length?showMembers(list,'依你的專業需求，這些目前顯示中的成員較相關：'):showFallback()}if(recommendByProblem(q))return;const list=findProfession(q);if(list.length)return showMembers(list,'依你的描述，這些目前顯示中的成員可能相關：');showFallback()}
  function build(){if(document.getElementById('buildingAssistant'))return;document.body.insertAdjacentHTML('beforeend',`<button class="building-assistant-launcher" id="assistantLauncher">問建築小幫手</button><div class="building-assistant" id="buildingAssistant"><div class="assistant-head"><div><b>砌好厝3.0｜建築小幫手</b><small>找人、找專業、解決工程問題</small></div><button class="assistant-close" id="assistantClose">×</button></div><div class="assistant-body" id="assistantBody"></div><div class="assistant-input"><input id="assistantInput" placeholder="直接描述你的問題…"><button id="assistantSend">送出</button></div></div>`);const panel=document.getElementById('buildingAssistant');document.getElementById('assistantLauncher').onclick=()=>{panel.classList.add('open');track('chat_open');if(!document.getElementById('assistantBody').children.length){addMsg('嗨，我是砌好厝3.0建築小幫手。你可以直接告訴我遇到的問題，或選擇你需要的服務。');showChoices()}};document.getElementById('assistantClose').onclick=()=>panel.classList.remove('open');document.getElementById('assistantSend').onclick=()=>{const i=document.getElementById('assistantInput');handleText(i.value);i.value=''};document.getElementById('assistantInput').addEventListener('keydown',e=>{if(e.key==='Enter'){e.preventDefault();document.getElementById('assistantSend').click()}});document.getElementById('assistantBody').addEventListener('click',e=>{const ib=e.target.closest('[data-intent]');if(ib){handleIntent(ib.dataset.intent);return}const nb=e.target.closest('[data-need]');if(nb){handleNeed(nb.dataset.need);return}const mb=e.target.closest('[data-member]');if(mb){track('chat_member_open',{member_name:mb.dataset.member,intent:currentIntent});if(typeof openProfile==='function')openProfile(mb.dataset.member)}});document.addEventListener('click',e=>{const contact=e.target.closest('.contact-action');if(contact){const name=document.getElementById('profileName')?.textContent?.trim()||'';track('contact_click',{member_name:name,meta:{label:contact.querySelector('.contact-type')?.textContent||''}})}const member=e.target.closest('.directory-card,.flow-person');if(member){const name=member.querySelector('.name')?.textContent?.trim()||'';if(name)track('member_card_click',{member_name:name})}})}
  window.addEventListener('DOMContentLoaded',()=>setTimeout(build,100));
})();
</script>
