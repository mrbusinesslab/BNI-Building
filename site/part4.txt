function closeProfile(options={}){
  document.getElementById('profileOverlay').classList.remove('open');
  document.body.style.overflow='';
  document.title='砌好厝3.0｜工程資源中心';
  if(options.updateUrl===false)return;
  const url=new URL(window.location.href);
  if(!url.searchParams.has('member'))return;
  if(window.history.state?.bniMemberOverlay){window.history.back();return}
  url.searchParams.delete('member');
  window.history.replaceState(null,'',url.pathname+url.search+url.hash);
}

}
async function shareProfileLink(){
  const url=profileShareUrl();
  const name=document.getElementById('profileName')?.textContent?.trim()||'專業成員';
  const btn=document.getElementById('shareProfileBtn');
  if(navigator.share){
    try{await navigator.share({title:name+'｜砌好厝3.0',text:'查看'+name+'的專業服務與聯絡方式',url});return}
    catch(e){if(e.name==='AbortError')return}
  }
  let copied=false;
  try{await navigator.clipboard.writeText(url);copied=true}catch(e){
    const area=document.createElement('textarea');area.value=url;area.style.position='fixed';area.style.opacity='0';document.body.appendChild(area);area.select();
    try{copied=document.execCommand('copy')}catch(err){}area.remove();
  }
  if(btn){btn.textContent=copied?'已複製連結':'請複製網址列連結';setTimeout(()=>{btn.textContent='分享連結'},2200)}
}
